local alphabet =
    "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"

local bits = data:gsub(".", function(c)
    if c == "=" then
        return ""
    end

    local n = alphabet:find(c, 1, true)
    if not n then
        error("Invalid encoded data")
    end

    n = n - 1
    local out = ""

    for i = 6, 1, -1 do
        out = out .. (
            n % 2^i - n % 2^(i-1) > 0
            and "1" or "0"
        )
    end

    return out
end)

local decoded = bits:gsub("%d%d%d?%d?%d?%d?%d?%d?", function(byte)
    if #byte ~= 8 then
        return ""
    end

    local n = 0

    for i = 1, 8 do
        if byte:sub(i, i) == "1" then
            n = n + 2^(8-i)
        end
    end

    return string.char(n)
end)

local fn, err = loadstring(decoded)

if not fn then
    error(err)
end

return fn()
